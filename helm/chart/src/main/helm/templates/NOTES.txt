###############################################################################
# Access the API gateway
###############################################################################
{{- if .Values.apiGateway.ingress.enabled }}
  The API gateway is exposed through an Ingress:

    http://{{ .Values.apiGateway.ingress.hostname }}{{ .Values.apiGateway.ingress.path }}

{{- else if .Values.apiGateway.route.enabled }}
  The API gateway is exposed through a Gateway API HTTPRoute:
{{- range .Values.apiGateway.route.hostnames }}
    http://{{ . }}
{{- end }}

{{- else }}
  The API gateway is not exposed externally. Forward its service port locally:

    kubectl --namespace {{ include "common.names.namespace" $ }} port-forward svc/{{ include "axon-showcase.api-gateway.fullname" $ }} 8080:8080

  Then call the API:

    curl http://localhost:8080/showcases
{{- end }}

###############################################################################
# Verify the deployment
###############################################################################
  Watch all pods become ready:

    kubectl --namespace {{ include "common.names.namespace" $ }} get pods -w -l "app.kubernetes.io/name={{ include "common.names.name" $ }}"

  Check the API gateway health (management port {{ .Values.apiGateway.containerPorts.managementServer }}):

    kubectl --namespace {{ include "common.names.namespace" $ }} port-forward svc/{{ include "axon-showcase.api-gateway.fullname" $ }} 8888:8888
    curl http://localhost:8888/actuator/health
